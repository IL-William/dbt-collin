# What collin is known to give otherwise than lineage-library says, one line
# of its expected.txt a mark:
#
#   project: missing: line    in expected.txt, and collin does not give it
#   project: extra: line      given by collin, and not in expected.txt
#
# A mark that stops holding fails the test, so that a fix removes it. Each
# names its limit above it.

# A scalar subquery in a select list. Correlated, the engine gives the outer
# side of the correlation as the column's value, and neither the column the
# subquery aggregates nor that outer side as a read; uncorrelated, the column
# is lost.
library: missing: book_ratings.best_stars <- raw.reviews.stars aggregate
library: missing: book_ratings reads books.book_id filter
library: extra: book_ratings.best_stars <- books.book_id aggregate
library: missing: book_ratings.last_review_at <- raw.reviews.written_at aggregate
library: extra: book_ratings.last_review_at lost
